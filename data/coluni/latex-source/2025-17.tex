\textbf{COLUNI 2025 -- Questão 17}\par\medskip Com base nos conjuntos $A=\{x\in\mathbb{N}/x\text{ é par}\}$, $B=\{y\in\mathbb{Z}/y\text{ é múltiplo de 3}\}$ e $C=\{z\in\mathbb{Q}/z\text{ é não ímpar}\}$, é CORRETO afirmar que:
\begin{enumerate}
\item $C\subset A$.
\item $A\cap B=\varnothing$.
\item $B\subset A$.
\item $A\subset C$.
\end{enumerate}
