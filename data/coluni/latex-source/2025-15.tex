\textbf{COLUNI 2025 -- Questão 15}\par\medskip Simplificando a expressão numérica
\[
Z=\frac{\left\{\dfrac{17}{8}-\left[0,2\cdot100+\left(\sqrt{49}+(-2)^5\right)\right]\right\}^{\frac12}}{\left(\dfrac18\right)^{-\frac13}\cdot\left(-\dfrac{3,6}{0,36}+\dfrac{194}{8}\right)^{\frac12}},
\]
obtém-se:
\begin{enumerate}
\item $Z=\dfrac{\sqrt{2}}{4}$.
\item $Z=\dfrac{2}{\sqrt{2}}$.
\item $Z=\dfrac{\sqrt{114}}{4}$.
\item $Z=\dfrac{2}{\sqrt{114}}$.
\end{enumerate}
